Note: autoplay only works in browsers when the video is also muted, so autoplay="true" implies muted. */}} {{- $src := .Get "src" | default (.Get 0) -}} {{- $poster := .Get "poster" -}} {{- $caption := .Get "caption" -}} {{- $autoplay := eq (.Get "autoplay" | default "false") "true" -}} {{- $loop := eq (.Get "loop" | default "false") "true" -}} {{- $url := $src -}} {{- with .Page.Resources.GetMatch $src -}} {{- $url = .RelPermalink -}} {{- end -}} {{- $posterUrl := $poster -}} {{- with $poster -}} {{- with $.Page.Resources.GetMatch . -}} {{- $posterUrl = .RelPermalink -}} {{- end -}} {{- end -}} {{- $type := printf "video/%s" (strings.TrimPrefix "." (path.Ext $url)) -}}

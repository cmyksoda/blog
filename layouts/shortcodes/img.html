{{/* Image shortcode with optional caption, width and alignment. {{< img src="photo.jpg" alt="A cat" >}} {{< img src="photo.jpg" alt="A cat" caption="Beans, helping" >}} {{< img src="photo.jpg" alt="A cat" width="50%" >}} {{< img src="photo.jpg" alt="A cat" width="300" align="center" >}} src — a file next to your index.md, or a path under static/ like "/images/photo.jpg", or a full https:// URL alt — description for screen readers. Please always write one. caption — optional text underneath width — "50%", "300px", or just "300" (treated as pixels) align — "center", "left" or "right" (default: left) Big images are resized down to 1400px automatically, so you can drop a full-size phone photo straight in. */}} {{- $src := .Get "src" | default (.Get 0) -}} {{- $alt := .Get "alt" | default "" -}} {{- $caption := .Get "caption" -}} {{- $width := .Get "width" -}} {{- $align := .Get "align" -}} {{- $maxWidth := 1400 -}} {{- $url := $src -}} {{- $natW := 0 -}} {{- $natH := 0 -}} {{- $found := false -}} {{- with .Page.Resources.GetMatch $src -}} {{- $found = true -}} {{- $sub := .MediaType.SubType -}} {{- if or (eq $sub "svg") (eq $sub "gif") -}} {{/* don't resize these — it breaks vectors / drops animation */}} {{- $url = .RelPermalink -}} {{- else -}} {{- $img := . -}} {{- if gt .Width $maxWidth -}} {{- $img = .Resize (printf "%dx" $maxWidth) -}} {{- end -}} {{- $url = $img.RelPermalink -}} {{- $natW = $img.Width -}} {{- $natH = $img.Height -}} {{- end -}} {{- end -}} {{/* Warn at build time if the file simply isn't there, rather than silently producing a broken image. Paths starting with / or http are meant to live elsewhere, so those are left alone. */}} {{- if and (not $found) (not (or (hasPrefix $src "/") (hasPrefix $src "http") (hasPrefix $src "data:"))) -}} {{- warnf "img shortcode: couldn't find %q next to %s — is the file in that folder, and spelled exactly right (capitals matter)?" $src .Page.File.Path -}} {{- end -}} {{/* Normalise a bare number like "300" into "300px" */}} {{- if and $width (findRE `^[0-9]+$` $width) -}} {{- $width = printf "%spx" $width -}} {{- end -}} {{- $class := slice "img-figure" -}} {{- with $align }}{{ $class = $class | append (printf "img-%s" .) }}{{ end -}} {{- if $width }}{{ $class = $class | append "img-sized" }}{{ end -}}
